-- Cho phép kéo thả 1 frame bằng cách giữ trên handle
local function makeDraggable(frame, handle)
    local dragging, dragStart, startPos = false, nil, nil
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = input.Position
            startPos  = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
end

-- Chỉ số → chữ cái: 1=A, 2=B, ... 27=AA
local function indexToLetter(i)
    local s = ""
    while i > 0 do
        local r = (i - 1) % 26
        s = string.char(65 + r) .. s
        i = math.floor((i - 1) / 26)
    end
    return s
end

-- Copy text ra clipboard (đa số executor hỗ trợ 1 trong các hàm sau)
local function copyToClipboard(t)
    local fn = (setclipboard or toclipboard or writeclipboard
        or (syn and syn.write_clipboard) or (Clipboard and Clipboard.set))
    if fn then
        local ok = pcall(fn, t)
        return ok
    end
    return false
end

local function findByPath(root, path)
    local cur = root
    for seg in string.gmatch(path, "[^.]+") do
        if cur then cur = cur:FindFirstChild(seg) else return nil end
    end
    return cur
end

local function isGloballyVisible(obj)
    if not obj then return false end
    local cur = obj
    while cur and cur ~= playerGui do
        if cur:IsA("GuiObject") and not cur.Visible then return false end
        cur = cur.Parent
    end
    return true
end

local function safeClick(button)
    if not (button and button.Visible) then return end
    local oldSel = GuiService.SelectedObject
    GuiService.SelectedObject = button
    task.wait(0.01)
    local clicked = false
    pcall(function()
        if getconnections then
            for _, c in pairs(getconnections(button.MouseButton1Click)) do c:Fire(); clicked = true end
            for _, c in pairs(getconnections(button.MouseButton1Down)) do c:Fire() end
            for _, c in pairs(getconnections(button.Activated))        do c:Fire() end
        end
    end)
    if not clicked then
        pcall(function()
            if button:IsA("TextButton") or button:IsA("ImageButton") then button:SimulateClick() end
        end)
    end
    GuiService.SelectedObject = oldSel
end

local function forceClickObject(guiObject)
    if not guiObject then return end
    local absPos  = guiObject.AbsolutePosition
    local absSize = guiObject.AbsoluteSize
    local fx = absPos.X + (absSize.X / 2) + OFFSET_X
    local fy = absPos.Y + (absSize.Y / 2) + OFFSET_Y
    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(fx, fy, 0, true,  game, 1)
        task.wait(0.04)
        VirtualInputManager:SendMouseButtonEvent(fx, fy, 0, false, game, 1)
    end)
    pcall(function()
        if firesignal then
            if guiObject:IsA("GuiButton") then
                firesignal(guiObject.MouseButton1Down)
                firesignal(guiObject.MouseButton1Click)
                firesignal(guiObject.Activated)
                firesignal(guiObject.MouseButton1Up)
            end
            local fDown = {UserInputType = Enum.UserInputType.MouseButton1, UserInputState = Enum.UserInputState.Begin}
            local fUp   = {UserInputType = Enum.UserInputType.MouseButton1, UserInputState = Enum.UserInputState.End}
            firesignal(guiObject.InputBegan, fDown)
            task.wait(0.02)
            firesignal(guiObject.InputEnded, fUp)
        end
    end)
end

    end)
end

-- Tìm slot trong hotbar; trả về slot, elements, mainframe
local function getHotbarSlot(slotName)
    local mainHud = playerGui:FindFirstChild("MainHud")
    if not mainHud then return nil end
    local elements  = mainHud:FindFirstChild("Elements")
    local hotbar    = elements and elements:FindFirstChild("Hotbar")
    local mainframe = hotbar and hotbar:FindFirstChild("MainFrame")
    if not (elements and mainframe) then return nil, elements, mainframe end
    local slot = mainframe:FindFirstChild(slotName) or elements:FindFirstChild(slotName)
    if not slot then
        for _, c in ipairs(mainframe:GetChildren()) do
            if c:IsA("GuiObject") then slot = c break end
        end
    end
    return slot, elements, mainframe
end

-- Chỉ dịch chuyển bản th��n tới vị trí + khoá góc nhìn mặc định
local function teleportSelf(pos)
    local char = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local head = char and char:FindFirstChild("Head")
    if root then
        root.CFrame = CFrame.new(pos)
        localPlayer.CameraMode = Enum.CameraMode.LockFirstPerson
        task.wait(0.12)
        local cam = workspace.CurrentCamera
        if cam and head then
            cam.CFrame = CFrame.lookAt(head.Position, head.Position + CAMERA_LOOK)
        end
    end
end

-- Đặt 1 unit tại vị trí (1 lần thử)
local function deployUnitOnce(unitName, targetPos)
    if not isRunning then return false end
    local slot, elements, mainframe = getHotbarSlot(unitName)
    if not (slot and elements and mainframe) then
        warn("❌ Không tìm thấy slot: " .. unitName); return false
    end

    slot.Parent      = elements
    slot.AnchorPoint = Vector2.new(0.5, 0.5)
    slot.Position    = UDim2.new(0.5, 0, 0.5, 0)
    task.wait(STEP_DELAY)

    teleportSelf(targetPos)
    task.wait(STEP_DELAY)

    forceClickObject(slot)
    task.wait(STEP_DELAY)

    local placeBtn
    local deadline = os.clock() + 1.0
    repeat
        placeBtn = findByPath(playerGui, "MainHud.Elements.Hud.PlaceButton")
        if placeBtn and isGloballyVisible(placeBtn) then break end
        task.wait(0.1)
    until os.clock() >= deadline

    local placed = false
    if placeBtn then forceClickObject(placeBtn); placed = true
    else warn("❌ Không tìm thấy PlaceButton (" .. unitName .. ")") end

    task.wait(0.35)
    if slot and slot.Parent == elements then slot.Parent = mainframe end
    localPlayer.CameraMode = Enum.CameraMode.Classic
    task.wait(STEP_DELAY)
    return placed
end

-- Đặt unit có RETRY (tự thử lại nếu hụt)
local function deployUnit(unitName, targetPos, maxTries)
    maxTries = maxTries or 3
    for attempt = 1, maxTries do
        if not isRunning then return false end
        if deployUnitOnce(unitName, targetPos) then
            print(string.format("✅ Đặt Unit %s @ %s (lần %d)", unitName, tostring(targetPos), attempt))
            return true
        end
        warn(string.format("⚠️ Hụt Unit %s (lần %d/%d)", unitName, attempt, maxTries))
        task.wait(0.4)
    end
    return false
end

local function waitAndClick(path, timeoutSec)
    timeoutSec = timeoutSec or 90
    local label = path:match("[^.]+$")
    setStatus("Chờ: " .. label .. "...")
    local deadline = os.clock() + timeoutSec
    while isRunning and os.clock() < deadline do
        local btn = findByPath(playerGui, path)
        if btn and isGloballyVisible(btn) then
            task.wait(0.1)
            safeClick(btn)
            print("✅ Đã click: " .. label)
            return true
        end
        task.wait(0.3)
    end
    warn("⚠️ Timeout khi chờ: " .. label)
    return false
end

-- Chờ VÔ HẠN nút PlayAgain (không bao giờ tự tắt)
local function waitForPlayAgain(path)
    local label = path:match("[^.]+$")
    local waited = 0
    while isRunning do
        local btn = findByPath(playerGui, path)
        if btn and isGloballyVisible(btn) then
            task.wait(0.2)
            for _ = 1, 3 do
                if not isRunning then break end
                safeClick(btn)
                task.wait(0.25)
                if not (btn.Parent and isGloballyVisible(btn)) then break end
            end
            print("✅ Đã click: " .. label)
            return true
        end
        task.wait(0.5)
        waited = waited + 0.5
        if waited % 10 < 0.5 then
            setStatus(string.format("Chờ PlayAgain... (%ds)", math.floor(waited)))
        end
    end
    return false
end

-- ── TEAM / UPGRADE ──
local function isTeamMenuOpen()
    local menu = findByPath(playerGui, PATHS.TeamMenu)
    return menu ~= nil and menu.Visible == true
end
local function toggleTeamMenu()
    local btn = findByPath(playerGui, PATHS.TeamBtn)
    if btn then safeClick(btn) else warn("❌ Không tìm thấy nút Team!") end
end
local function closeTeamMenu()
    if not isTeamMenuOpen() then return end
    local closeBtn = findByPath(playerGui, PATHS.CloseTeam)
    if closeBtn then safeClick(closeBtn) else toggleTeamMenu() end
    task.wait(0.25)
end

-- Upgrade slot từ A→B trong durationSec; tự mở lại Team nếu bị đóng; xong tắt "X"
local function runUpgradeCycle(slotA, slotB, durationSec)
    if not isRunning then return end
    slotA = math.max(1, math.floor(tonumber(slotA) or 1))
    slotB = math.max(slotA, math.floor(tonumber(slotB) or slotA))
    durationSec = tonumber(durationSec) or 30
    setStatus(string.format("Upgrade slot %d→%d / %ds...", slotA, slotB, durationSec))

    if not isTeamMenuOpen() then toggleTeamMenu(); task.wait(0.5) end

    local deadline = os.clock() + durationSec
    while isRunning and os.clock() < deadline do
        if not isTeamMenuOpen() then toggleTeamMenu(); task.wait(0.25) end
        for i = slotA, slotB do
            if not isRunning then break end
            local upgradeBtn = findByPath(playerGui,
                "MainHud.Menus.TeamManager.Menu.Towers.Listings." .. i .. ".UpgradeButtons.Upgrade")
            if upgradeBtn then safeClick(upgradeBtn) end
        end
        task.wait(0.1)
    end
    task.wait(0.3)
    closeTeamMenu()
    print(string.format("✅ Xong upgrade slot %d→%d!", slotA, slotB))
end

local function defaultBlock(t)
    if t == "wait"          then return { type = t, ms = 1000 }
    elseif t == "teleport"  then return { type = t, posIndex = 1 }
    elseif t == "unit"      then return { type = t, unit = "5", posIndex = 1 }
    elseif t == "upgrade"   then return { type = t, from = 1, to = 4, dur = 31 }
    elseif t == "difficulty" then return { type = t, level = 3 }
    elseif t == "loop"      then return { type = t, count = 0 } -- 0 = vô hạn
    else return { type = t } end
end

-- Mô tả 1 khối dễ đọc (dùng cho phần "Tạo Mã")
local function describeBlock(b)
    if b.type == "wait" then
        return "⏱ Chờ " .. (tonumber(b.ms) or 0) .. " ms"
    elseif b.type == "teleport" then
        local p = positions[b.posIndex]
        return "🧍 Dịch chuyển → " .. ((p and p.label) or "?")
    elseif b.type == "unit" then
        local p = positions[b.posIndex]
        return "🎯 Đặt Unit ô " .. tostring(b.unit) .. " → " .. ((p and p.label) or "?")
    elseif b.type == "upgrade" then
        return string.format("⬆ Upgrade ô %s→%s trong %ss", tostring(b.from), tostring(b.to), tostring(b.dur))
    elseif b.type == "play" then
        return "🎮 Chơi (Play)"
    elseif b.type == "startgame" then
        return "🚀 Bắt đầu game"
    elseif b.type == "difficulty" then
        return "⚙ Chọn độ khó " .. tostring(b.level) .. ":" .. (DIFFICULTY_NAMES[b.level] or "")
    elseif b.type == "playagain" then
        return "🏳 Chơi lại"
    elseif b.type == "loop" then
        local c = tonumber(b.count) or 0
        return "🔁 Lặp lại từ đầu " .. (c <= 0 and "(vô hạn)" or (c .. " lần"))
    end
    return b.type
end

-- [H] EXECUTOR – chạy danh sách khối từ trên xuống
-- ============================================================
local function runOneBlock(b)
    if b.type == "wait" then
        local ms = tonumber(b.ms) or 0
        setStatus("Chờ " .. ms .. " ms")
        local left = ms / 1000
        while isRunning and left > 0 do
            local step = math.min(0.2, left)
            task.wait(step)
            left = left - step
        end
    elseif b.type == "teleport" then
        local p = positions[b.posIndex]
        if p then setStatus("Dịch chuyển → " .. p.label); teleportSelf(p.pos)
                  localPlayer.CameraMode = Enum.CameraMode.Classic end
    elseif b.type == "unit" then
        local p = positions[b.posIndex]
        if p then setStatus("Đặt Unit " .. b.unit .. " → " .. p.label)
                  deployUnit(b.unit, p.pos, 3) end
    elseif b.type == "upgrade" then
        runUpgradeCycle(b.from, b.to, b.dur)
    elseif b.type == "play" then
        waitAndClick(PATHS.Play, 180); task.wait(0.6)
    elseif b.type == "startgame" then
        waitAndClick(PATHS.StartGame, 180); task.wait(0.6)
    elseif b.type == "difficulty" then
        local path = DIFFICULTY_PATHS[b.level]
        if path then setStatus("Chọn độ khó " .. (DIFFICULTY_NAMES[b.level] or b.level))
                     waitAndClick(path, 90); task.wait(0.6) end
    elseif b.type == "playagain" then
        waitForPlayAgain(PATHS.PlayAgain); task.wait(1.0)
    end
end

-- Chạy theo chỉ số. Khối "loop" sẽ nhảy về đầu (vô hạn nếu count=0,
-- hoặc lặp count lần rồi đi tiếp).
local function runCompiledScript(blocks)
    local loopCount = {} -- đếm số lần đã lặp tại mỗi khối loop
    local i = 1
    while isRunning and i <= #blocks do
        local b = blocks[i]
        if b.type == "loop" then
            local target = tonumber(b.count) or 0
            if target <= 0 then
                setStatus("🔁 Lặp lại từ đầu (vô hạn)")
                i = 1
            else
                loopCount[i] = (loopCount[i] or 0) + 1
                if loopCount[i] < target then
                    setStatus(string.format("🔁 Lặp %d/%d", loopCount[i], target))
                    i = 1
                else
                    loopCount[i] = nil
                    i = i + 1
                end
            end
        else
            runOneBlock(b)
            i = i + 1
        end
    end
    pcall(function() localPlayer.CameraMode = Enum.CameraMode.Classic end)
end

local function renderPositions()
    for _, ch in ipairs(PosList:GetChildren()) do

    for i, p in ipairs(positions) do
        new("TextLabel", {
            Size = UDim2.new(1, 0, 0, 18), BackgroundTransparency = 1, LayoutOrder = i,
            Text = string.format("  %s  (%.0f, %.0f, %.0f)", p.label, p.pos.X, p.pos.Y, p.pos.Z),
            TextColor3 = C.text, Font = Enum.Font.Gotham, TextSize = 10,
            TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd,
        }, PosList)
    end
end

local function makeCycleBtn(parent, xOffset, width, text, onClick)
    local b = new("TextButton", {
        Size = UDim2.new(0, width, 0, 22), Position = UDim2.new(0, xOffset, 0.5, -11),
        BackgroundColor3 = Color3.fromRGB(255,255,255), Text = text,

    }, parent)
    corner(b, 5)

end

local function createBlockRow(block, index)
    local def = BLOCK_DEFS[block.type]
    local row = new("Frame", {
        Size = UDim2.new(1, 0, 0, 34), BackgroundColor3 = def.color,
        BorderSizePixel = 0, LayoutOrder = index,
    }, Canvas)
    corner(row, 7)

    -- Nhãn khối
    local labelW = (#def.params >= 3) and 96 or 130
    new("TextLabel", {
        Size = UDim2.new(0, labelW, 1, 0), Position = UDim2.new(0, 8, 0, 0),
        BackgroundTransparency = 1, Text = def.label, TextColor3 = C.text,
        Font = Enum.Font.GothamSemibold, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, row)

    -- Caption nhỏ phía trên 1 ô số
    local function captioned(xOffset, w, caption, value, onChange)
        new("TextLabel", {
            Size = UDim2.new(0, w, 0, 9), Position = UDim2.new(0, xOffset, 0, 2),
            BackgroundTransparency = 1, Text = caption, TextColor3 = C.text,
            Font = Enum.Font.Gotham, TextSize = 8, TextTransparency = 0.25,
        }, row)
        local box = new("TextBox", {
            Size = UDim2.new(0, w, 0, 20), Position = UDim2.new(0, xOffset, 0, 12),
            BackgroundColor3 = Color3.fromRGB(255,255,255), Text = value,
            TextColor3 = Color3.fromRGB(20,20,20), Font = Enum.Font.GothamSemibold, TextSize = 11,
            ClearTextOnFocus = false,
        }, row)
        corner(box, 5)
        box.FocusLost:Connect(function() onChange(box.Text) end)
        return box
    end
    local function numField(field, caption, w, x)
        captioned(x, w, caption, tostring(block[field]), function(t)
            local n = tonumber(t); if n then block[field] = n end
        end)
        return x + w + 4
    end

    local x = 8 + labelW + 6
    -- Tham số
    for _, p in ipairs(def.params) do
        if p == "ms" then
            x = numField("ms", "ms", 56, x)
        elseif p == "unit" then
            captioned(x, 36, "ô", tostring(block.unit), function(t)
                if t ~= "" then block.unit = t end
            end); x = x + 40
        elseif p == "from"  then x = numField("from", "ô từ", 40, x)
        elseif p == "to"    then x = numField("to", "đến", 40, x)
        elseif p == "dur"   then x = numField("dur", "giây", 44, x)
        elseif p == "count" then
            captioned(x, 56, "số lần (0=∞)", tostring(block.count), function(t)
                local n = tonumber(t); if n then block.count = math.max(0, math.floor(n)) end
            end); x = x + 60
        elseif p == "pos" then
            local btn
            btn = makeCycleBtn(row, x, 44, (positions[block.posIndex] and positions[block.posIndex].label) or "?",
            function()
                if #positions == 0 then return end
                block.posIndex = (block.posIndex % #positions) + 1
                btn.Text = positions[block.posIndex].label
            end)
            x = x + 50
        elseif p == "level" then
            local btn
            btn = makeCycleBtn(row, x, 100, block.level .. ":" .. DIFFICULTY_NAMES[block.level],
            function()
                block.level = (block.level % 5) + 1
                btn.Text = block.level .. ":" .. DIFFICULTY_NAMES[block.level]
            end)
            x = x + 106
        end
    end

    -- Nút điều khiển bên phải: ↑ ↓ ✕
    local function ctrlBtn(posX, txt, col, fn)
        local b = new("TextButton", {
            Size = UDim2.new(0, 22, 0, 22), Position = UDim2.new(1, posX, 0.5, -11),
            BackgroundColor3 = col, Text = txt, TextColor3 = C.text,
            Font = Enum.Font.GothamBold, TextSize = 12,
        }, row)
        corner(b, 5)
        b.MouseButton1Click:Connect(fn)
    end
    ctrlBtn(-78, "↑", Color3.fromRGB(0,0,0):Lerp(def.color,0.4), function()
        if index > 1 then
            scriptBlocks[index], scriptBlocks[index-1] = scriptBlocks[index-1], scriptBlocks[index]
            renderScript()
        end
    end)
    ctrlBtn(-52, "↓", Color3.fromRGB(0,0,0):Lerp(def.color,0.4), function()
        if index < #scriptBlocks then
            scriptBlocks[index], scriptBlocks[index+1] = scriptBlocks[index+1], scriptBlocks[index]
            renderScript()
        end
    end)
    ctrlBtn(-26, "✕", C.red, function()
        table.remove(scriptBlocks, index)
        renderScript()
    end)
end

renderScript = function()
    for _, ch in ipairs(Canvas:GetChildren()) do
        if ch:IsA("Frame") then ch:Destroy() end
    end
    if #scriptBlocks == 0 then
        new("TextLabel", {
            Size = UDim2.new(1, 0, 0, 40), BackgroundTransparency = 1,
            Text = "Bấm khối lệnh bên trái để thêm vào đây...",
            TextColor3 = C.muted, Font = Enum.Font.Gotham, TextSize = 12,
        }, Canvas)
        return
    end
    for i, b in ipairs(scriptBlocks) do
        createBlockRow(b, i)
    end
end

-- Trạng thái test
local testActive   = false
local testSlot     = nil
local testOrig     = nil  -- { parent=, pos=, anchor= }

local function startOffsetTest()
    local slot, elements = getHotbarSlot(OFFSET_TEST_SLOT)
    if not (slot and elements) then
        setStatus("Không tìm thấy unit ô " .. OFFSET_TEST_SLOT .. " (mở game trước nhé)")
        return false
    end
    testSlot = slot
    testOrig = { parent = slot.Parent, pos = slot.Position, anchor = slot.AnchorPoint }
    slot.Parent      = elements
    slot.AnchorPoint = Vector2.new(0.5, 0.5)
    slot.Position    = UDim2.new(0.5, 0, 0.5, 0)
    return true
end

local function stopOffsetTest()
    if testSlot and testOrig then
        pcall(function()
            testSlot.AnchorPoint = testOrig.anchor
            testSlot.Position    = testOrig.pos
            testSlot.Parent      = testOrig.parent
        end)
    end
    testSlot, testOrig = nil, nil
end

        local cy = ap.Y + sz.Y / 2 + OFFSET_Y

        offsetMarkerLbl.Text  = string.format(" Lệch: %d, %d", math.floor(OFFSET_X), math.floor(OFFSET_Y))
    end
end)

local function fmtVal(v)
    if type(v) == "string" then return string.format("%q", v) end
    if type(v) == "number" then
        if math.floor(v) == v then return tostring(math.floor(v)) end
        return tostring(v)
    end
    return tostring(v)
end

local function makeDraggable(frame, handle)
    local dragging, dragStart, startPos = false, nil, nil
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos = true, input.Position, frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,

    if not obj then return false end

local function safeClick(button)
    if not (button and button.Visible) then return end
    local oldSel = GuiService.SelectedObject
    GuiService.SelectedObject = button
    task.wait(0.01)
    local clicked = false
    pcall(function()
        if getconnections then
            for _, c in pairs(getconnections(button.MouseButton1Click)) do c:Fire(); clicked = true end
            for _, c in pairs(getconnections(button.MouseButton1Down)) do c:Fire() end
            for _, c in pairs(getconnections(button.Activated))        do c:Fire() end
        end
    end)
    if not clicked then
        pcall(function()
            if button:IsA("TextButton") or button:IsA("ImageButton") then button:SimulateClick() end
        end)
    end
    GuiService.SelectedObject = oldSel
end
local function forceClickObject(guiObject)
    if not guiObject then return end
    local absPos, absSize = guiObject.AbsolutePosition, guiObject.AbsoluteSize
    local fx = absPos.X + (absSize.X / 2) + OFFSET_X
    local fy = absPos.Y + (absSize.Y / 2) + OFFSET_Y
    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(fx, fy, 0, true,  game, 1)
        task.wait(0.04)
        VirtualInputManager:SendMouseButtonEvent(fx, fy, 0, false, game, 1)
    end)
    pcall(function()
        if firesignal then
            if guiObject:IsA("GuiButton") then
                firesignal(guiObject.MouseButton1Down); firesignal(guiObject.MouseButton1Click)
                firesignal(guiObject.Activated);        firesignal(guiObject.MouseButton1Up)
            end
            local fDown = {UserInputType = Enum.UserInputType.MouseButton1, UserInputState = Enum.UserInputState.Begin}
            local fUp   = {UserInputType = Enum.UserInputType.MouseButton1, UserInputState = Enum.UserInputState.End}

    if not slot then

local function deployUnit(unitName, targetPos, maxTries)
    maxTries = maxTries or 3
    for attempt = 1, maxTries do
        if not isRunning then return false end
        if deployUnitOnce(unitName, targetPos) then
            print(string.format("OK Dat Unit %s (lan %d)", unitName, attempt)); return true
        end
        warn(string.format("Hut Unit %s (lan %d/%d)", unitName, attempt, maxTries))
        task.wait(0.4)
    end
    return false
end
local function waitAndClick(path, timeoutSec)
    timeoutSec = timeoutSec or 90
    local label = path:match("[^.]+$")
    setStatus("Cho: " .. label .. "...")
    local deadline = os.clock() + timeoutSec
    while isRunning and os.clock() < deadline do
        local btn = findByPath(playerGui, path)
        if btn and isGloballyVisible(btn) then
            task.wait(0.1); safeClick(btn); print("OK click: " .. label); return true
        end
        task.wait(0.3)
    end
    warn("Timeout cho: " .. label); return false
end

local function runUpgradeCycle(slotA, slotB, durationSec)
    if not isRunning then return end
    slotA = math.max(1, math.floor(tonumber(slotA) or 1))
    slotB = math.max(slotA, math.floor(tonumber(slotB) or slotA))
    durationSec = tonumber(durationSec) or 30
    setStatus(string.format("Upgrade slot %d->%d / %ds...", slotA, slotB, durationSec))
    if not isTeamMenuOpen() then toggleTeamMenu(); task.wait(0.5) end
    local deadline = os.clock() + durationSec
    while isRunning and os.clock() < deadline do
        if not isTeamMenuOpen() then toggleTeamMenu(); task.wait(0.25) end
        for i = slotA, slotB do
            if not isRunning then break end

                "MainHud.Menus.TeamManager.Menu.Towers.Listings." .. i .. ".UpgradeButtons.Upgrade")

    print(string.format("OK xong upgrade slot %d->%d!", slotA, slotB))
end

local function runOneBlock(b)
    if b.type == "wait" then
        local ms = tonumber(b.ms) or 0
        setStatus("Cho " .. ms .. " ms")
        local left = ms / 1000
        while isRunning and left > 0 do
            local step = math.min(0.2, left); task.wait(step); left = left - step
        end
    elseif b.type == "teleport" then
        local p = positions[b.posIndex]
        if p then setStatus("Dich chuyen -> " .. p.label); teleportSelf(p.pos)
                  localPlayer.CameraMode = Enum.CameraMode.Classic end
    elseif b.type == "unit" then
        local p = positions[b.posIndex]
        if p then setStatus("Dat Unit " .. b.unit .. " -> " .. p.label); deployUnit(b.unit, p.pos, 3) end
    elseif b.type == "upgrade" then
        runUpgradeCycle(b.from, b.to, b.dur)
    elseif b.type == "play" then
        waitAndClick(PATHS.Play, 180); task.wait(0.6)
    elseif b.type == "startgame" then
        waitAndClick(PATHS.StartGame, 180); task.wait(0.6)
    elseif b.type == "difficulty" then
        local path = DIFFICULTY_PATHS[b.level]
        if path then setStatus("Chon do kho " .. (DIFFICULTY_NAMES[b.level] or b.level))
                     waitAndClick(path, 90); task.wait(0.6) end
    elseif b.type == "playagain" then
        waitForPlayAgain(PATHS.PlayAgain); task.wait(1.0)
    end
end
local function runCompiledScript(blocks)
    local loopCount = {}
    local i = 1
    while isRunning and i <= #blocks do
        local b = blocks[i]
        if b.type == "loop" then
            local target = tonumber(b.count) or 0
            if target <= 0 then setStatus("Lap lai tu dau (vo han)"); i = 1
            else
                loopCount[i] = (loopCount[i] or 0) + 1
                if loopCount[i] < target then
                    setStatus(string.format("Lap %d/%d", loopCount[i], target)); i = 1
                else loopCount[i] = nil; i = i + 1 end
            end
        else runOneBlock(b); i = i + 1 end
    end
    pcall(function() localPlayer.CameraMode = Enum.CameraMode.Classic end)
end

-- [O] TẠO SCRIPT → MENU KÍCH HOẠT RIÊNG
-- ============================================================
local function buildRunMenu()
    -- Bản sao chương trình tại thời điểm bấm "Tạo Script"
    local snapshot = {}
    for _, b in ipairs(scriptBlocks) do
        local copy = {}
        for k, v in pairs(b) do copy[k] = v end
        snapshot[#snapshot + 1] = copy
    end

    -- Xóa menu cũ
    for _, g in ipairs({CoreGui, playerGui}) do
        local old = g:FindFirstChild("FullMacroRunGui")
        if old then old:Destroy() end
    end

    local RunGui = new("ScreenGui", { Name = "FullMacroRunGui", ResetOnSpawn = false })
    pcall(function() RunGui.Parent = CoreGui end)
    if not RunGui.Parent then RunGui.Parent = playerGui end

    local rf = new("Frame", {
        Size = UDim2.new(0, 240, 0, 150), Position = UDim2.new(0.5, 160, 0.5, -75),
        BackgroundColor3 = C.bg, BorderSizePixel = 0,
    }, RunGui)
    corner(rf, 12)
    stroke(rf, C.panel2, 1)

    local rbar = new("Frame", {
        Size = UDim2.new(1, 0, 0, 30), BackgroundColor3 = C.green, BorderSizePixel = 0,
    }, rf)
    corner(rbar, 12)
    new("TextLabel", {
        Size = UDim2.new(1, -40, 1, 0), Position = UDim2.new(0, 10, 0, 0),
        BackgroundTransparency = 1, Text = "▶ SCRIPT (" .. #snapshot .. " khối)",
        TextColor3 = C.text, Font = Enum.Font.GothamBold, TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, rbar)
    local rClose = new("TextButton", {
        Size = UDim2.new(0, 24, 0, 20), Position = UDim2.new(1, -30, 0.5, -10),
        BackgroundColor3 = C.red, Text = "✕", TextColor3 = C.text,
        Font = Enum.Font.GothamBold, TextSize = 12,
    }, rbar)
    corner(rClose, 5)
    makeDraggable(rf, rbar)

    local rStatus = new("TextLabel", {
        Size = UDim2.new(1, -16, 0, 16), Position = UDim2.new(0, 8, 0, 36),
        BackgroundTransparency = 1, Text = "Trạng thái: Sẵn sàng", TextColor3 = C.muted,
        Font = Enum.Font.Gotham, TextSize = 10, TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, rf)

    new("TextLabel", {
        Size = UDim2.new(1, -16, 0, 14), Position = UDim2.new(0, 8, 0, 54),
        BackgroundTransparency = 1, Text = "Lặp: dùng khối 🔁 trong chương trình",
        TextColor3 = C.muted, Font = Enum.Font.Gotham, TextSize = 9,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, rf)

    local runThread
    local actBtn = new("TextButton", {
        Size = UDim2.new(1, -16, 0, 30), Position = UDim2.new(0, 8, 0, 72),
        BackgroundColor3 = C.title, Text = "⚡ KÍCH HOẠT", TextColor3 = C.text,
        Font = Enum.Font.GothamBold, TextSize = 13,
    }, rf)
    corner(actBtn, 7)
    local stopBtn = new("TextButton", {
        Size = UDim2.new(1, -16, 0, 24), Position = UDim2.new(0, 8, 0, 106),
        BackgroundColor3 = C.red, Text = "⏹ DỪNG", TextColor3 = C.text,
        Font = Enum.Font.GothamSemibold, TextSize = 11,
    }, rf)
    corner(stopBtn, 6)

    local function stopRun()
        isRunning = false
        if runThread then pcall(function() task.cancel(runThread) end) runThread = nil end
        pcall(function() localPlayer.CameraMode = Enum.CameraMode.Classic end)
        actBtn.Text = "⚡ KÍCH HOẠT"
        actBtn.BackgroundColor3 = C.title
    end

    actBtn.MouseButton1Click:Connect(function()
        if isRunning then return end
        if #snapshot == 0 then rStatus.Text = "Trạng thái: Chưa có khối nào!" return end
        isRunning = true
        activeStatusLabel = rStatus
        actBtn.Text = "⏳ ĐANG CHẠY..."
        actBtn.BackgroundColor3 = Color3.fromRGB(120, 95, 0)
        runThread = task.spawn(function()
            runCompiledScript(snapshot)
            stopRun()
            setStatus("Hoàn tất.")
        end)
    end)
    stopBtn.MouseButton1Click:Connect(function()
        stopRun()
        rStatus.Text = "Trạng thái: Đã dừng."
    end)
    rClose.MouseButton1Click:Connect(function()
        stopRun()
        RunGui:Destroy()
    end)

    print("✅ Đã tạo script với " .. #snapshot .. " khối lệnh!")
end

CreateBtn.MouseButton1Click:Connect(buildRunMenu)
ClearBtn.MouseButton1Click:Connect(function()
    scriptBlocks = {}
    renderScript()
end)
