local function StopFly()
    Settings.IsFlying = false
    if Settings.Mover then Settings.Mover:Destroy(); Settings.Mover = nil end
    if Settings.Aligner then Settings.Aligner:Destroy(); Settings.Aligner = nil end
    if Settings.Attach then Settings.Attach:Destroy(); Settings.Attach = nil end
    
    local Character = Player.Character
    if Character then
        local Hum = Character:FindFirstChildOfClass("Humanoid")
        if Hum then Hum:ChangeState(Enum.HumanoidStateType.GettingUp) end
    end
end

local function StartFly()
    if Settings.IsFlying then return end
    local Character = Player.Character
    if not Character then return end
    local Root = Character:FindFirstChild("HumanoidRootPart")
    local Hum = Character:FindFirstChildOfClass("Humanoid")
    if not Root or not Hum then return end

    Settings.IsFlying = true
    -- Tắt vật lý đi bộ, đổi state để không bị rớt
    Hum:ChangeState(Enum.HumanoidStateType.Physics) 
    
    -- TẠO CÔNG NGHỆ MỚI (CONSTRAINTS)
    local Attach = Instance.new("Attachment", Root)
    Settings.Attach = Attach
    
    local LinearVel = Instance.new("LinearVelocity", Root)
    LinearVel.Attachment0 = Attach
    LinearVel.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
    LinearVel.RelativeTo = Enum.ActuatorRelativeTo.World
    LinearVel.MaxForce = math.huge
    LinearVel.VectorVelocity = Vector3.new(0, 0, 0)
    Settings.Mover = LinearVel
    
    local AlignOri = Instance.new("AlignOrientation", Root)
    AlignOri.Mode = Enum.OrientationAlignmentMode.OneAttachment
    AlignOri.Attachment0 = Attach
    AlignOri.MaxTorque = math.huge
    AlignOri.Responsiveness = 200 -- Tốc độ xoay cực nhanh và mượt
    AlignOri.CFrame = Root.CFrame
    Settings.Aligner = AlignOri

    task.spawn(function()
        while Settings.FlyEnabled and Settings.IsFlying and Character and Character.Parent and Hum.Health > 0 do
            RunService.RenderStepped:Wait()
            local Cam = workspace.CurrentCamera
            
            -- Lấy Joystick Vector chuẩn chỉnh từ bạn!
            local moveVector = ControlModule:GetMoveVector()
            
            if moveVector.Magnitude > 0 then
                -- Bay chuẩn 3D
                local flyDir = (Cam.CFrame.RightVector * moveVector.X) + (Cam.CFrame.LookVector * -moveVector.Z)
                LinearVel.VectorVelocity = flyDir.Unit * Settings.FlySpeed
                
                -- Xoay kiểu Superman
                AlignOri.CFrame = Cam.CFrame
            else
                -- Hover mượt mà
                LinearVel.VectorVelocity = Vector3.new(0, 0.1, 0) -- Giữ trên không nhẹ nhàng
                
                -- Fix UX: Nhìn thẳng theo Camera nhưng không gật gù
                local camLook = Cam.CFrame.LookVector
                AlignOri.CFrame = CFrame.lookAt(Root.Position, Root.Position + Vector3.new(camLook.X, 0, camLook.Z))
            end
        end
        StopFly()
    end)
end

local function teleportToNextPoint()

    if isTeleporting then
        return
    end

    isTeleporting = true

    local character = LocalPlayer.Character

    if not character then
        isTeleporting = false
        return
    end

    local humanoidRootPart = character:FindFirstChild("HumanoidRootPart")

    local camera = Workspace.CurrentCamera

    if humanoidRootPart and camera then

        -- Lấy vị trí hiện tại
        local targetPosition = PATH_POSITIONS[currentPoint]

        -- Dịch chuyển
        humanoidRootPart.CFrame = CFrame.new(targetPosition)

        -- Chờ ổn định
        task.wait(0.05)

        -- Khóa góc nhìn first person
        LocalPlayer.CameraMode = Enum.CameraMode.LockFirstPerson

        -- Ép góc nhìn camera
        local camPos = camera.CFrame.Position

        camera.CFrame = CFrame.lookAt(
            camPos,
            camPos + CAMERA_LOOK
        )

        -- Chuyển sang điểm tiếp theo
        currentPoint += 1

        -- Quay vòng lại point 1
        if currentPoint > #PATH_POSITIONS then
            currentPoint = 1
        end

        -- Update text
        updateButtonText()
    end

    isTeleporting = false
end

-- [MỚI] Bảng lưu trữ thời gian Cooldown để chống Spam Click
local clickCooldowns = {}
local COOLDOWN_TIME = 2.5 -- Đợi 2.5 giây sau khi ấn mới được ấn lại NÚT ĐÓ

local function safeClick(button)
    if button and button.Visible and button.Active then
        local guiService = game:GetService("GuiService")
        local oldSelected = guiService.SelectedObject
        
        guiService.SelectedObject = button
        task.wait(0.01)
        
        local clicked = false
        
        pcall(function()
            if getconnections then
                for _, connection in pairs(getconnections(button.MouseButton1Click)) do connection:Fire() clicked = true end
                for _, connection in pairs(getconnections(button.MouseButton1Down)) do connection:Fire() clicked = true end
                for _, connection in pairs(getconnections(button.Activated)) do connection:Fire() clicked = true end
            end
        end)
        
        if not clicked then
            pcall(function()
                if button:IsA("TextButton") or button:IsA("ImageButton") then
                    button:SimulateClick()
                    clicked = true
                end
            end)
        end
        
        if not clicked then
            pcall(function()
                local virtualInput = game:GetService("VirtualInputManager")
                virtualInput:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
                virtualInput:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
            end)
        end
        
        guiService.SelectedObject = oldSelected
    end
end

local function findButtonByPath(path)
    local current = playerGui
    for name in string.gmatch(path, "[^.]+") do
        if current then
            current = current:FindFirstChild(name)
        else
            return nil
        end
    end
    return current
end

local function isButtonGloballyVisible(button)
    if not button then return false end
    
    local current = button
    while current and current ~= playerGui do
        if current:IsA("GuiObject") and not current.Visible then
            return false
        end
        current = current.Parent
    end
    return true
end

local function checkAndClickAll()
    local currentTime = os.clock() -- Lấy thời gian hiện tại của hệ thống (chính xác hơn tick)
    
    for name, path in pairs(buttonPaths) do
        local button = findButtonByPath(path)
        
        if button and isButtonGloballyVisible(button) then
            -- [MỚI] Kiểm tra Cooldown: Nếu chưa từng click, hoặc thời gian trôi qua đã lớn hơn COOLDOWN_TIME
            if not clickCooldowns[name] or (currentTime - clickCooldowns[name]) >= COOLDOWN_TIME then
                
                clickCooldowns[name] = currentTime -- Ghi nhận lại thời gian vừa click
                safeClick(button)
                print("[Auto Farm] Đã ấn: " .. name)
                
                task.wait(0.25) -- Nghỉ 1 nhịp siêu nhỏ giữa các nút khác nhau
            end
        end
    end
end

local function setupListeners()
    for name, path in pairs(buttonPaths) do
        local button = findButtonByPath(path)
        if button then
            if not button:GetAttribute("AutoClickConnected") then
                -- CHỈ lắng nghe sự kiện Visible (Ẩn/Hiện), gỡ bỏ .Changed để chống lag
                button:GetPropertyChangedSignal("Visible"):Connect(checkAndClickAll)
                button:SetAttribute("AutoClickConnected", true)
            end
        end
    end
end

task.spawn(setupListeners)

playerGui.DescendantAdded:Connect(function(descendant)

        if descendant.Name == lastName then
            task.wait(0.1)
            setupListeners()
            checkAndClickAll()
            break
        end
    end
end)

-- DRAG SYSTEM: CHỈ KÉO Ở TITLE BAR
--------------------------------------------------

local dragging = false
local dragStart
local startPos
local dragInput

local function updateDrag(input)
    local delta = input.Position - dragStart
    local newPos = UDim2.new(
        startPos.X.Scale,
        startPos.X.Offset + delta.X,
        startPos.Y.Scale,
        startPos.Y.Offset + delta.Y
    )
    MainFrame.Position = newPos
end

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        dragInput = input

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
                dragInput = nil
            end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and input == dragInput then
        updateDrag(input)
    end
end)
